\section{梯度流耦合与质量}\label{sec:couplingandmass}

\subsection{梯度流耦合}

$\langle E(t)\rangle$ 的重正化群不变性允许我们固定标度 $\mu=\sqrt{\rho/t}$
（$\rho\in\mathbb{R}$ 为常数），而它与 $\alphas$ 的比例关系启发我们定义一个
``梯度流耦合''
\begin{equation}
  \begin{split}
    \alphagf(\mu) \equiv \frac{32\pi\rho^2}{3\NA\mu^4}\,\langle
    E(\rho/\mu^2)\rangle \equiv
    \alphas(\mu)\sum_{n=0}^\infty\left(\frac{\alphas(\mu)}{4\pi}\right)^n
    e_{n}(\rho)\,.
    %% \label{eq:}
  \end{split}
\end{equation}
对通常的梯度流耦合定义\,\cite{Luscher:2010iy}，$\rho$ 取 $1/8=0.125$；而从微
扰的角度看，$\rho=1/(2e^{\EulerGamma})=0.281\ldots$
似乎更为合适（参见 \noeqn{eq:mu0}）。

对 $\alphas$ 微扰求解 \noeqn{eq:et} 得到
\begin{equation}
  \begin{split}
    \alphas = \alphagf\left[ 1 - e_{1}(\rho)\frac{\alphagf}{4\pi} +
      (2e_{1}^2(\rho)-e_{2}(\rho))\left(\frac{\alphagf}{4\pi}\right)^2 +
      \ldots\right]\,,
    %% \label{eq:}
  \end{split}
\end{equation}
系数 $e_{n}(\rho)$ 由 \noeqn{eq:efull} 给出。这类质量无关的重正化方案之间的
变换保持 $\beta$ 函数前两个系数不变，即
\begin{equation}
  \begin{split}
    \mu^2\dderiv{}{\mu^2}\alphagf(\mu) &= \alphagf(\mu)\betagf{}(\alphagf)\,,\\
    \betagf{}(\alphagf) = -\sum_{n=0}^\infty\betagf{n}
    \left(\frac{\alphagf}{4\pi}\right)^n
    &= - \frac{\alphagf}{4\pi}\beta_0
     - \left(\frac{\alphagf}{4\pi}\right)^2\beta_1
     -\sum_{n=2}^\infty\betagf{n}
     \left(\frac{\alphagf}{4\pi}\right)^n\,,
    %% \label{eq:}
  \end{split}
\end{equation}
其中 $\beta_0$ 与 $\beta_1$ 取自 \noeqn{eq:anombg}。第三个系数为
\begin{equation}
  \begin{split}
    \betagf{2} = \beta_2 - e_{1}(\rho)\,\beta_1 + (e_{2}(\rho)
    - e_{1}^2(\rho))\,\beta_0\,,
    %% \label{eq:}
  \end{split}
\end{equation}
其中 \msbar\ 系数 $\beta_2$ 我们针对 SU(3) 规范群引用如下：
\begin{equation}
  \begin{split}
    \beta_2 = \frac{2857}{2} - \frac{5033}{18}\NF + \frac{325}{54}\NF^2\,.
    %% \label{eq:}
  \end{split}
\end{equation}

\subsection{梯度流质量}

类似地，我们可以定义梯度流夸克质量
\begin{equation}
  \begin{split}
    \massgf_f(\mu) = -\frac{8\pi^2 \rho}{\NC\mu^2}m_f\dderiv{\langle
      \mathring{S}(\rho/\mu^2)\rangle}{m_f}\Big|_{m=0} = m_f(\mu)\sum_{n=0}^\infty
    \hat{s}_{n}(\rho)\left(\frac{\alphagf(\mu)}{4\pi}\right)^n\,,
    %% \label{eq:}
  \end{split}
\end{equation}
其中
\begin{equation}
  \begin{split}
    \hat{s}_{0}(\rho) = s_{0}(\rho)\,,\qquad
    \hat{s}_{1}(\rho) = s_{1}(\rho)\,,\qquad
    \hat{s}_{2}(\rho) = s_{2}(\rho) - s_{1}(\rho)\, e_{1}(\rho)\,,
    %% \label{eq:}
  \end{split}
\end{equation}
各 $s_{n}$ 与 $e_{1}$ 见 \noeqn{eq:sfull} 与 \noeqn{eq:efull}。逆关系为
\begin{equation}
  \begin{split}
    m_f = \massgf_f\left[1-\hat{s}_{1}(\rho)\frac{\alphagf}{4\pi} +
      (\hat{s}_{1}^2(\rho)-\hat{s}_{2}(\rho))
      \left(\frac{\alphagf}{4\pi}\right)^2
      + \ldots\right]\,.
    %% \label{eq:}
  \end{split}
\end{equation}
新质量满足重正化群方程
\begin{equation}
  \begin{split}
    \mu\dderiv{}{\mu}\massgf_f(\mu) = \gammagf_m(\alphagf)\,\massgf_f(\mu)
   \equiv - \massgf_f(\mu)\,\sum_{n=0}^\infty
    \left(\frac{\alphagf(\mu)}{4\pi}\right)^{n+1}\gammagf_{m,n}
    \label{eq:rgme}
  \end{split}
\end{equation}
其中
\begin{equation}
  \begin{split}
    \gammagf_{m,0} &= \gamma_{m,0}\,,\qquad
    \gammagf_{m,1} = \gamma_{m,1}
    - \gamma_{m,0}\, e_{1}(\rho)
    + 2\,\beta_0\,\hat{s}_{1}(\rho)
    \,,\\
    \gammagf_{m,2} &=
    \gamma_{m,2}
    - 2\,\gamma_{m,1}\,e_{1}(\rho)
    + \gamma_{m,0}\,\left[2\,e_{1}^2(\rho) - e_{2}(\rho)\right]\\&\qquad
    + 2\,\beta_1\hat{s}_{1}(\rho)
    + 2\,\beta_0\,\left(2\,\hat{s}_{2}(\rho)-\hat{s}^2_1(\rho)\right)\,,
    %% \label{eq:}
  \end{split}
\end{equation}
其中 $\beta_0$、$\beta_1$、$\gamma_{m,0}$、$\gamma_{m,1}$ 见
\noeqn{eq:anombg}，而三圈 \msbar\ 系数 $\gamma_{m,2}$ 针对 SU(3) 规范群引用如
下：
\begin{equation}
  \begin{split}
    \gamma_{m,2} = 2498 -
    \left(\frac{4432}{27}+\frac{320}{3}\zeta(3)\right) \NF -
    \frac{280}{81}\,\NF^2\,.
    %% \label{eq:}
  \end{split}
\end{equation}
