\chapter{Simulation Engines and Signal Enhancement}

Between Wilson's action and a physics number stands machinery: an
algorithm that samples gauge fields, an improved fermion discretization,
and smearing techniques that make hadron signals visible at the large
momenta demanded by parton physics. The four papers of this chapter are
the load-bearing members of that machinery; two more (the Wuppertal
smearing of G\"usken et al.\ 1989 and the clover term of
Sheikholeslami--Wohlert 1985) are used so pervasively by the library's
ensembles that they are acknowledged here without separate entries.

% ----------------------------------------------------------------------
\section{Hybrid Monte Carlo}
\papercard{S.\ Duane, A.\ D.\ Kennedy, B.\ J.\ Pendleton, D.\ Roweth}
{Hybrid Monte Carlo}
{Phys.\ Lett.\ B \textbf{195}, 216--222 (1987)}
{No arXiv (preprint era) | DOI: 10.1016/0370-2693(87)90011-7}
\whyselected{HMC is the engine behind essentially every dynamical ensemble
used in this library (MILC HISQ, CLS, clover ensembles of CLQCD, domain
wall). It is cited explicitly by the HMC auto-tuning paper found in the
collection, by both flow-based-sampling papers, and by the charmed-meson
ensemble paper.}
\physics{HMC marries molecular dynamics to Metropolis. Momenta $p$ are
attached to the field variables; the system is integrated along a
fictitious Hamiltonian trajectory of length $\tau$,
\begin{equation}
  H = \frac{1}{2}p^2 + S[U], \qquad \dot U = i\!\left[\frac{\partial H}{\partial p}\right]\!U ,
\end{equation}
and the endpoint is accepted with probability $\min(1,e^{-\Delta H})$.
Because the integrator is reversible and area-preserving, detailed balance
is exact while the global proposal suppresses random-walk behavior:
decorrelation grows like $\tau$ rather than like step count. The cost ---
repeated Dirac inversions --- drove thirty years of solver innovation,
culminating in the multigrid and mixed-precision algorithms on which
libraries such as QUDA (used by this corpus) are built.}
\impact{Every ensemble in every lattice PDF calculation of this library
was generated by HMC or a direct descendant (Hasenbusch mass
preconditioning, multiple time scales). The AI-sampling line of
Chapter~8 defines itself as an alternative proposal layer on top of the
same Metropolis logic --- and cites this paper as its baseline.}
\cardend

\section{Analytic Link Smearing}
\papercard{C.\ Morningstar and M.\ Peardon}
{Analytic smearing of $SU(3)$ link variables in lattice QCD}
{Phys.\ Rev.\ D \textbf{69}, 054501 (2004)}
{arXiv:hep-lat/0307022 | DOI: 10.1103/PhysRevD.69.054501}
\whyselected{Stout smearing is cited throughout the gradient-flow chapters
of this library as the discrete ancestor of the flow, and appears in the
reference lists of L\"uscher's own papers ($\sim$7/99 among the flow
corpus). It is also the standard noise-reduction tool in the smeared-quasi-PDF
calculations.}
\physics{Each thin link is replaced by a smoothed link obtained by one
exponential step toward the staple force,
\begin{equation}
  U_\mu(n)\;\to\; e^{\,\mathrm{i}\rho\,\Omega_\mu(n)}\,U_\mu(n),\qquad
  \Omega_\mu = \mathrm{antiherm}\Big(\sum_{\pm\nu\neq\mu}\!U_\nu(n)U_\mu(n+\hat\nu)U^\dagger_\nu(n+\hat\mu)\Big),
\end{equation}
iterated a few times. The update is analytic in the links, gauge covariant
by construction, and cheap. The connection to the gradient flow is
intimate: stout smearing with weight $\rho$ is precisely the Euler
discretization of the flow equation (\ref{eq:flow}) of Chapter~3 with flow
time $t=\rho$ per step.}
\impact{The equivalence ``stout = coarse-grained flow'' is how
L\"uscher~2010 motivates the physical meaning of the flow; Monahan and
Orginos exploit the same structure when they smear quasi-PDF operators
with the continuous flow (Chapter~6). Within the library it is also the
workhorse smearing for nucleon-structure matrix elements.}
\cardend

\section{Distillation}
\papercard{M.\ Peardon, J.\ Bulava, J.\ Foley, C.\ Morningstar, J.\ Juge,
A.\ J\"uttner, K.\ Orginos, et al.\ (Hadron Spectrum Collaboration)}
{A novel quark-field creation operator construction for hadronic physics in lattice QCD}
{Phys.\ Rev.\ D \textbf{80}, 054506 (2009)}
{arXiv:0905.2160 | DOI: 10.1103/PhysRevD.80.054506}
\whyselected{An original article of this library, and the enabling
technology of the entire HadStruc pseudo-PDF program contained in it
(gluon PDFs, gluon helicity, high-momentum distillation): cited by at
least 4 articles of the corpus as method source.}
\physics{Instead of Jacobi smearing applied ad hoc at source or sink,
distillation fixes once and for all a low-mode projector built from the
three-dimensional Laplacian on each time slice,
\begin{equation}
  \square^{[N]}(t)=\sum_{k=1}^{N}\lvert v_k(t)\rangle\langle v_k(t)\rvert,\qquad
  -\square_3\,v_k=\lambda_k v_k,
\end{equation}
and writes every quark propagator through the \emph{perambulator}
$\tau$, a small object computed once per configuration:
$\langle x|e^{-Ht}|y\rangle = \Phi(x,t)\,\tau(t)\,\Phi^\dagger(y,0)$.
Any number of interpolators can then be combined into variational
matrices \emph{post hoc}, with no new Dirac solves. The variational
method of L\"uscher--Wolff becomes affordable at scale.}
\impact{In this library, Egerer et al.'s \emph{Distillation at High
Momentum} extends the idea with momentum phases to reach nucleon momenta
above 3 GeV --- exactly the regime where the pseudo-ITD gluon calculations
(Khan et al., Egerer et al.) operate. The HadStruc gluon-PDF line is
inconceivable without this paper.}
\cardend

\section{Momentum Smearing}
\papercard{G.\ S.\ Bali, B.\ Lang, B.\ U.\ Musch, A.\ Sch\"afer}
{Novel quark smearing for hadrons with high momenta in lattice QCD}
{Phys.\ Rev.\ D \textbf{93}, 094515 (2016)}
{arXiv:1602.05525 | DOI: 10.1103/PhysRevD.93.094515}
\whyselected{The standard signal-enhancement tool of every boosted-hadron
calculation in this corpus: quoted by the LaMET helicity, transversity,
deuteron-like dibaryon and kinematically-enhanced-operator papers
($\sim$6/99).}
\physics{Ordinary spatial smearing optimizes overlap with the \emph{rest-frame}
wave function; a boosted hadron's wave function oscillates, so overlap dies
like powers of $P^z$. Momentum smearing multiplies the smearing kernel by a
plane-wave phase tuned to the target momentum,
\begin{equation}
  \tilde\psi(\mathbf{x}) = \sum_{\mathbf{y}} e^{\,\mathrm{i}\mathbf{P}\cdot(\mathbf{x}-\mathbf{y})}\,
  K(\mathbf{x}-\mathbf{y})\,\psi(\mathbf{y}),
\end{equation}
so the smeared operator tracks the moving state. Signal quality for
three-point functions at $P^z\sim1$--$2$ GeV improves severalfold ---
often the difference between a PDF extraction and no extraction.}
\impact{Combined with distillation in the library's HadStruc papers, and
with optimized interpolator bases in Zhang et al.'s kinematically-enhanced
operators paper (which reports further order-of-magnitude gains from
$P^2$-enhanced overlaps), this technique defines the current art of
high-momentum matrix elements.}
\cardend
