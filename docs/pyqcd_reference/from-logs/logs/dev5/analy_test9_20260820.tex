\end{abstract}

\subsection{A5 关键细节与全链物理公式完整推导}
\label{sec:a5}
本节为物理核心，按“是什么 $\rightarrow$ 怎么工作 $\rightarrow$ 为什么这样设计”三层展开，每公式编号可 \eqref{eq:*} 引用。

\subsubsection{A5.1 梯度流：Wilson flow 与 RK3}
连续流方程（Luescher 2010 \texttt{\nolinkurl{\_gradient\_flow.py:7-11}}）：
\begin{equation}
\partial_t B_\mu(t,x) = D_\nu G_{\nu\mu}(t,x),\quad B_\mu(0,x)=A_\mu(x),\quad G_{\mu\nu}= \partial_\mu B_\nu-\partial_\nu B_\mu+[B_\mu,B_\nu]
\label{eq:flow_cont}
\end{equation}
格点离散化：$V_\mu(x,t)$ 为流规范场，Wilson 作用量导数由 6-staple 构造
\begin{equation}
\Omega_\mu(x)=\sum_{\nu\ne\mu}\left[U_\nu(x)U_\mu(x+\hat\nu)U^\dagger_\nu(x+\hat\mu)+U^\dagger_\nu(x-\hat\nu)U_\mu(x-\hat\nu)U_\nu(x-\hat\nu+\hat\mu)\right]
\label{eq:staple}
\end{equation}
（\texttt{\nolinkurl{\_gradient\_flow.py:62-104}} 整场预 roll 加速 1.5 倍，逐位一致）。流生成元
\begin{equation}
Z(V)(x,\mu)=P_{ah}\left[\Omega_\mu(x)V^\dagger_\mu(x)\right],\quad P_{ah}[M]=\frac{M-M^\dagger}{2}-\frac{\mathrm{Tr}(M-M^\dagger)}{2N}\mathbf{1}
\label{eq:Zdef}
\end{equation}
投影到 $\mathfrak{su}(3)$ 保持 $SU(3)$（\texttt{\nolinkurl{\_gradient\_flow.py:111-122}}）。时间积分用三阶 Runge--Kutta（Luescher Eq.(3.8)，\texttt{\nolinkurl{\_gradient\_flow.py:14-20}}）：
\begin{align}
W_0 &= V_t,\quad Z_0=Z(W_0) \nonumber\\
W_1 &= e^{\epsilon Z_0/4}W_0,\quad Z_1=Z(W_1) \nonumber\\
W_2 &= e^{(8\epsilon/9)Z_1-(17\epsilon/36)Z_0}W_1,\quad Z_2=Z(W_2) \label{eq:RK3}\\
V_{t+\epsilon} &= e^{(3\epsilon/4)Z_2-(8\epsilon/9)Z_1+(17\epsilon/36)Z_0}W_2 \nonumber
\end{align}
指数用截断级数 $e^A\simeq I+A+A^2/2+A^3/6+A^4/24$ $+$ 投影（\texttt{\nolinkurl{\_gradient\_flow.py:46-54}}，$\epsilon=0.05$ 时 $O(\epsilon^3)$ 精度，60 步到 $t=3$）。流观测量
\begin{equation}
E(t)=\frac14 G^{a}_{\mu\nu}G^{a}_{\mu\nu}\simeq \frac14\sum_{\mu<\nu}\mathrm{Tr} F_{\mu\nu}F^\dagger_{\mu\nu},\quad t^2\langle E(t)\rangle=0.3 \text{ 定 }t_0
\label{eq:Et0}
\end{equation}
（\texttt{\nolinkurl{\_gradient\_flow.py:162-200}}）。\textbf{量纲校验：} $[t]=L^2$，$[E]=M^4$，$t^2E$ 无量纲；$t\to0$ 回 $A_\mu$，$t\to\infty$ 趋平滑，单调 $E(t)$ 递减为物理自洽信号（见 A9）。

\subsubsection{A5.2 胶子场强与对偶（Clover 构造）}
plaquette $P_{\mu\nu}(x)=U_\mu(x)U_\nu(x+\hat\mu)U^\dagger_\mu(x+\hat\nu)U^\dagger_\nu(x)$，Clover 平均四叶（donghx 算法，\texttt{\nolinkurl{\_gluon\_ope.py:63-115}}）：
\begin{equation}
F_{\mu\nu}(x)= -\frac{i}{8}\sum_{k=1}^4\left(P^{(k)}_{\mu\nu}-P^{(k)\dagger}_{\mu\nu}\right)+O(a^2),\quad F_{\mu\nu}=-F_{\nu\mu},\;F^\dagger_{\mu\nu}=-F_{\mu\nu}
\label{eq:clover}
\end{equation}
误差 $O(a^2)$ 因对称平均消除 $O(a)$。对偶张量（\texttt{\nolinkurl{\_gluon\_ope.py:40-60}}，$\mathcal{T}_{\mu\nu\rho\sigma}=\frac12\epsilon_{\mu\nu\rho\sigma}$）：
\begin{equation}
\tilde F_{\mu\nu}= \frac12\sum_{\rho\sigma}\epsilon_{\mu\nu\rho\sigma}F_{\rho\sigma},\quad \epsilon_{0123}=+1
\label{eq:dual}
\end{equation}
系数由 \texttt{\nolinkurl{build\_tensor4}} 查表实现，$F_{\tilde F}$ 同后端（numpy/torch）。

\subsubsection{A5.3 Wilson 线与 staple}
直 Wilson 线（\texttt{\nolinkurl{\_gluon\_ope.py:135-185}} roll 乘积）：
\begin{equation}
W(z\hat z;0)= \mathcal{P}\exp\left(ig\int_0^z A_z(s\hat z)ds\right)\simeq \prod_{k=0}^{z-1}U_z(x+k\hat z),\quad W^\dagger W=1+O(a^2)
\label{eq:Wline}
\end{equation}
staple Wilson 线（TMD 必需，\texttt{\nolinkurl{renorm/\_tmd.py:52-92}} Eq.:staple\_wilson）：
\begin{equation}
W_\sqsubset(z,b_\perp)=U^\dagger_z((z+L)\hat z+b_\perp\hat b,b_\perp\hat b)\cdot U_\perp((z+L)\hat z+b_\perp,(z+L)\hat z)\cdot U_z((z+L)\hat z,z\hat z)
\label{eq:stapleW}
\end{equation}
$L$ 为 staple 臂长（默认 $L=z$，\texttt{\nolinkurl{\_tmd.py:69-70}}）。b=0 时 $W_\sqsubset\to W(z)$，直观退化检验通过。

\subsubsection{A5.4 可乘性重整化组合 $O(z,b_\perp)$}
基础表示矩阵元（\texttt{\nolinkurl{\_tmd.py:123-150}} Eq.:M\_fund\_tmd）：
\begin{equation}
\mathcal{M}^{\mu\lambda;\nu\rho}(z,b_\perp)=\sum_{\mathbf{x}}\mathrm{Tr}\left[F^{\mu\lambda}(\mathbf{x}+z\hat z+b_\perp\hat b)W_\sqsubset(z,b_\perp)F^{\nu\rho}(\mathbf{x})W^\dagger_\sqsubset(z,b_\perp)\right]
\label{eq:Mfund}
\end{equation}
求和 $\sum_{\mathbf{x}}$ 为空间平均（保平移不变）。一圈计算表明 $\mathcal{M}$ 的 UV 发散矩阵 $3\times3$ 非对角（$tx,ty,xy$ 混），但组合
\begin{equation}
O(z,b_\perp)=\mathcal{M}^{tx;tx}(z,b_\perp)+\mathcal{M}^{ty;ty}(z,b_\perp)-2\mathcal{M}^{xy;xy}(z,b_\perp)
\label{eq:Odef}
\end{equation}
（\texttt{\nolinkurl{\_tmd.py:152-161}} Eq.:O\_mult\_tmd）使反对称部分精确抵消，剩余发散乘法可重整（至少一圈，\texttt{\nolinkurl{\_tmd.py:15-17}} 注释）。\textbf{对称性论证：} $t\leftrightarrow x$ 交换下 $F_{tx}$ 与 $F_{yz}$ 对偶，$O$ 属 $2^{++}$ 牵引表示，避免与标量混；$b_\perp\to0$ 时 $\mathcal{M}^{xy}\propto b_\perp^2\to0$，$O\to$ 准 PDF 算符，连续极限自洽。

不变振幅（\texttt{\nolinkurl{\_tmd.py:20-22}} Eq.:Mpp\_tmd）：
\begin{equation}
\mathcal{M}^{ti;it}(z,b_\perp)+\mathcal{M}^{ji;ij}(z,b_\perp)=2p_0^2\,\mathcal{M}_{pp}(\nu,b_\perp),\quad \nu=zP_z
\label{eq:Mpp}
\end{equation}
其中 $\nu$ 为 Ioffe 时间，无量纲。光锥关系（\texttt{\nolinkurl{\_tmd.py:22-23}}）：
\begin{equation}
-\mathcal{M}_{pp}(\nu,b_\perp)=\frac12\int_{-1}^1 dx\,e^{-ix\nu}\,x\,g(x,b_\perp)
\label{eq:MppPDF}
\end{equation}
傅里叶反演即得 $xg(x,b_\perp)$（\texttt{\nolinkurl{\_tmd.py:240-272}} \texttt{\nolinkurl{invariant\_amplitude}}）。

\subsubsection{A5.5 核子 2pt 与不相连比值（含谱分解推导）}
质子 2pt（\texttt{\nolinkurl{pipeline/\_steps.py:200-350}} 蒸馏，\texttt{\nolinkurl{\_tmd9.py:76-81}} 调度）：
\begin{equation}
C_2(\mathbf{P},t)=\sum_{\mathbf{x}}e^{-i\mathbf{P}\cdot\mathbf{x}}\langle\chi_p(\mathbf{x},t)\bar\chi_p(0)\rangle = \sum_n |Z_n(\mathbf{P})|^2 e^{-E_n(\mathbf{P})t}
\label{eq:C2spec}
\end{equation}
基态 $E_0(\mathbf{P})=\sqrt{M^2+\mathbf{P}^2}+O(a^2)$（色散见图）。不相连 TMD 3pt 因子化（\texttt{\nolinkurl{\_tmd\_ratio.py:5-17}}）：
\begin{equation}
C_3(dt,d\tau,z,b)=C_2(dt)\,{\rm OPE}(d\tau,z,b),\quad {\rm OPE}(d\tau,z,b)=\sum_{\mathbf{x}}O(z,b;d\tau,\mathbf{x})
\label{eq:C3fact}
\end{equation}
由 \texttt{\nolinkurl{tmd\_matrix\_elements\_time}} 逐 $t$ 空间求和 $(n_z,n_b,N_t)$ 提供（\texttt{\nolinkurl{\_tmd.py:184-202}}）。真空扣除与比值（\texttt{\nolinkurl{\_tmd\_ratio.py:76-112}} code\_1 算法）：
\begin{equation}
C_3^{\rm disc}=C_3-C_2\langle{\rm OPE}\rangle,\quad R(dt,d\tau,z,b)=\big\langle C_3^{\rm disc}/C_2\big\rangle_{t_i}
\label{eq:ratio}
\end{equation}
$\langle\cdot\rangle_{t_i}$ 为源时间平均（$NT=72$，\texttt{\nolinkurl{\_tmd\_ratio.py:79-84}} roll 实现）。两态拟合模型（\texttt{\nolinkurl{\_disconnected.model\_ratio:1-30}}）：
\begin{equation}
R(dt,d\tau)=c_0(z,b)+c_1 e^{-dE\,d\tau}+c_1 e^{-dE\,(dt-d\tau)},\quad dE=E_1-E_0
\label{eq:ratioModel}
\end{equation}
由 \eqref{eq:C2spec} 两态展开严格导出（基态 $+$ 第一激发态，振幅 $c_1$ 对称）。10 组态时 bootstrap 200 样本满秩（\texttt{\nolinkurl{\_tmd\_ratio.py:63-66}}），$N_{\rm data}=14$ 时条件数 $>10^8$ 回退对角近似（\texttt{\nolinkurl{\_tmd\_ratio.py:163-167}}）。

\subsubsection{A5.6 自重整化 $Z_R$ 与混合方案}
$Z_R$ 参数化（arXiv:2510.17758 Eq.3-8，\texttt{\nolinkurl{renorm/\_zr.py:1-60}}）：
\begin{equation}
Z_R(z,a,\mu)=\exp\left[\frac{kz}{a\ln(a\Lambda)}+\frac12\ln\!\left(1+\frac{d}{\ln(a\Lambda)}\right)^2+\frac{5C_A}{3b_0}\ln\frac{\ln(1/a\Lambda)}{\ln(\mu/\Lambda)}+(m_0+m_2a^2)z+fa+f_2a^2\right]
\label{eq:ZR}
\end{equation}
逐项物理：$kz/(a\ln)$ 线性发散（含对数修正）、第二项 $\ln$ 重求和、第三项 5CA/3b0 的 MS-bar 匹配、$m_0z$ 质量重整化、$fa$ 离散化。$Z_{\rm MS}$（\texttt{\nolinkurl{\_zr.py:20-30}}）：
\begin{equation}
Z_{\rm MS}(z,\mu)=1+\frac{\alpha_s C_A}{4\pi}\left[\frac53\ln\frac{z^2\mu^2}{4e^{-2\gamma_E}}+3\right]
\label{eq:ZMS}
\end{equation}
test9 采用梯度流自重整比值简化版（\texttt{\nolinkurl{\_tmd9.py:216-224}}）$h_R(z,b)=c_0(z,b)/c_0(0,b)$，长距用混合拼接（\texttt{\nolinkurl{\_hybrid.py:1-40}}）：
\begin{equation}
h_R(z,P_z)=\begin{cases} h_B(z,P_z)/h_B(z,0), & z<z_s \\ h_B(z,P_z)/Z_R(z)\cdot\eta_s, & z\ge z_s \end{cases},\quad \eta_s=Z_R(z_s)/h_B(z_s,0)
\label{eq:hybrid}
\end{equation}
保证 $z_s$ 连续。$\lambda=zP_z$ 外推（\texttt{\nolinkurl{\_hybrid.py:30-50}}）：
\begin{equation}
h_R(\lambda)\sim l_1\lambda^{-a_1}e^{-\lambda/\lambda_0},\quad \lambda\to\infty
\label{eq:lambdaExtrap}
\end{equation}
压制傅里叶截断振荡。

\subsubsection{A5.7 准 TMD 与 NLO 匹配（含 SFTX）}
准 TMD（\texttt{\nolinkurl{\_tmdextract.py:7-9}} Eq.:quasi\_tmd）：
\begin{equation}
x\tilde g(x,b_\perp,P_z)=\frac1{\mathcal{N}}\int\frac{dz}{2\pi P_z}e^{-ixzP_z}h_R(z,b_\perp,P_z),\quad \mathcal{N}=P_z^2/P_t^2
\label{eq:quasiTMD}
\end{equation}
$P_t=\sqrt{M^2+P_z^2}$，余弦离散化（\texttt{\nolinkurl{\_tmdextract.py:60-69}}）。NLO 匹配核（\texttt{\nolinkurl{renorm/\_matching.py:30-80}}，zengch 逻辑，1 圈）：
\begin{align}
g_1(\xi)&=-\frac{2(1-\xi+\xi^2)^2}{1-\xi}\ln\frac{\xi}{\xi-1}-\frac{11-28\xi+18\xi^2-12\xi^3}{6(1-\xi)},\quad \xi<0 \nonumber\\
g_2(\xi)&=\frac{2(1-\xi+\xi^2)^2}{1-\xi}\left[-\ln\frac{\mu^2}{4y^2P_z^2}+\ln\xi(1-\xi)\right]-\frac{15-56\xi+102\xi^2-96\xi^3+48\xi^4}{6(1-\xi)},\;0<\xi<1 \nonumber\\
g_3(\xi)&=\frac{2(1-\xi+\xi^2)^2}{1-\xi}\ln\frac{\xi}{\xi-1}+\frac{11-28\xi+18\xi^2-12\xi^3}{6(1-\xi)},\quad \xi>1 \label{eq:g123}\\
g_0(\xi,y)&=\frac56\left[-\frac1{|1-\xi|}+\frac{2\,{\rm Si}((1-\xi)|y|\lambda_s)}{\pi(1-\xi)}\right],\quad {\rm Si}(x)=\int_0^x\frac{\sin t}{t}dt \nonumber
\end{align}
分区 $g_{xy}= -g_0-g_1$ ($\xi<0$)、$g_0+g_2$ ($0<\xi<1$)、$g_0+g_3$ ($\xi>1$)（\texttt{\nolinkurl{\_matching.py:80-110}}）。匹配矩阵
\begin{equation}
Z_{ij}=\delta_{ij}+\frac{\alpha_s C_A}{2\pi}M_{ij},\;M_{ij}=\frac{g_{ij}}{y_j}-\delta_{ij}y_i\sum_k\frac{g_{ik}}{y_k^2},\quad h_R^{\rm PDF}=Z^{-1}h_R
\label{eq:Zij}
\end{equation}
（\texttt{\nolinkurl{\_matching.py:110-140}}）。TMD 混合匹配（\texttt{\nolinkurl{\_tmdextract.py:104-176}}）同构，但加入快度演化与软函数：
\begin{equation}
x\tilde g(x,b)=\int\frac{dy}{|y|}C^{\rm hybrid}(x/y,\lambda_s,\mu/yP_z)\frac{y\,g(y,b)}{\sqrt{S_I(b)}}e^{\frac12\ln(\zeta_z/\zeta)K(b)}
\label{eq:TMDmatch}
\end{equation}
$K(b)$ 为 CS 核，$S_I$ 内禀软函数，离散实现 \texttt{\nolinkurl{tmd\_matching\_hybrid}} 的 $Z_{ij}$ 同 \eqref{eq:Zij} 结构，$A_s=\alpha_s/4\pi$ 约定（\texttt{\nolinkurl{\_tmdextract.py:147-152}}）。

SFTX 小流时展开（Suzuki 2013，\texttt{\nolinkurl{\_tmdextract.py:187-217}}）：
\begin{equation}
\mathcal{O}_{\overline{\rm MS}}(\mu)=\left[1+\frac{\alpha_s}{4\pi}c(t,\mu)\right]\mathcal{O}_{\rm flow}(t),\;c(t,\mu)=2b_0[\ln(2\mu^2t)+\gamma_E]
\label{eq:SFTX}
\end{equation}
$b_0=11-2N_f/3$，$\gamma_E$ 欧拉常数。

\section{结论与局限}
\begin{conclbox}
\textbf{核心结论：} \texttt{\nolinkurl{tag:test9}} 系列在 10 组态真实数据上端到端验证梯度流重整化核子胶子 TMD-PDF 的可行性：Wilson flow（\eqref{eq:flow_cont}--\eqref{eq:RK3}）自动有限化、staple 组合 \eqref{eq:Odef} 一圈可乘化、不相连比值 \eqref{eq:ratio}--\eqref{eq:ratioModel} 稳健提取 $h_B$、比值重整化 $\rightarrow$ 傅里叶 \eqref{eq:quasiTMD} $\rightarrow$ NLO 匹配 \eqref{eq:Zij}--\eqref{eq:TMDmatch} 得 $xg(x,b)$ 与 $K(b)$，全链与 \texttt{\nolinkurl{pure\_pyqcd}} 的 C1--C5 理论预期一致；三视角中主体结构六层、关系主链 4 条、思路“先保真再加速后拓量”均经代码与日志确证。
\end{conclbox}
\begin{refbox}
\texttt{\nolinkurl{docs/analy\_pyqcd\_20260818.pdf}} 与 \texttt{\nolinkurl{docs/pure\_pyqcd\_20260818.pdf}} 的“梯度流 $+$ 自重整化 $+$ 混合 $+$ NLO”四段论为本报告提供理论背书；\texttt{\nolinkurl{refer/papers/gluon\_tmd\_gradient\_flow\_continuum.tex:Eq.O\_mult\_tmd}} 直接给出 \eqref{eq:Odef} 的文献源。
\end{refbox}
\begin{warnbox}
\textbf{局限与风险（\textcolor{warn}{未验证} 项如实列出）：} 10 组态统计不足（$c_0$ 大 $z,b$ 噪声 $>100\%$、$K$ $\pm0.6$）；长距混合未拟合真 $Z_R$（\textcolor{warn}{未验证}）；高阶 $O(\alpha_s^2)$ 与 $O(a^2)$ 离散化、连续/手征/体积外推均未做（\texttt{\nolinkurl{\_extrapolate.py}} 待接入）；仅 $24^3\times72$ 单系综、单流时间 $\tau=3a^2$，$t$ 依赖残留 \textcolor{warn}{未验证}；手工图表曾现 4 类错误（已修但提示需回归测试固化）。
\end{warnbox}
